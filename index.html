Código Google Apps Script (Extensiones → Apps Script en tu Sheet) 
function doPost(e) {
  var ss = SpreadsheetApp.openById("14HQcFP1zY52tNY2JgV-1mrvrGDtjS97tLt0ZaVK6uc4");
  var data = JSON.parse(e.postData.contents);
  var sheetName = data.sheetName; // Ej: "Quincena 30-11-2026"
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) {
    sheet = ss.insertSheet(sheetName);
  } else {
    sheet.clear();
  }
  sheet.appendRow(["Categoría", "Producto", "Cantidad / Peso", "Precio ($)", "Estado", "Notas"]);
  sheet.getRange(1, 1, 1, 6).setFontWeight("bold").setBackground("#065f46").setFontColor("#ffffff");
  data.items.forEach(function(item) {
    sheet.appendRow([item.category, item.name, item.qty, item.price, "Comprado", item.note || ""]);
  });
  sheet.appendRow(["", "", "TOTAL QUINCENA", data.total, "Tope: $150.00", ""]);
  return ContentService.createTextOutput(JSON.stringify({status: "ok", sheet: sheetName})).setMimeType(ContentService.MimeType.JSON);
}
